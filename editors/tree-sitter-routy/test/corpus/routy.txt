POST /orders/{shop} {
  handler: orders.Create
  params {
    email: "a@b.c"
    password
  }
  only: [dev, staging]
  confirm: true
  timeout: 10s
  headers {
    Authorization: "Bearer ${Login().body.token}"
    Idempotency-Key: uuid()
  }
  body { email, items: [{ sku: "A-1", qty: 2 }], }
  poll body.status == "paid" every 1s for 30s
  expect {
    status == 201
    headers.content-type.contains("json")
    body matches Order
    body.id matches /^\d+$/i
  }
  save order_id = body.id
}
