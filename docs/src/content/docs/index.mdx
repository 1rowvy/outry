import { Card, CardGrid } from '@astrojs/starlight/components';

```sh
curl -fsSL https://raw.githubusercontent.com/1rowvy/routy/master/install.sh | sh
```

```http title="api/users/create.http"
POST {{base}}/users
Authorization: Bearer {{token}}

{"name": "Viktor"}

> save user_id = body.id
> assert status == 201
```

<CardGrid>
	<Card title="Requests live in git" icon="document">
		One `.http` file per request. Review them in pull requests, grep them, edit them in any editor —
		the app picks up changes instantly.
	</Card>
	<Card title="Same engine everywhere" icon="laptop">
		The desktop app and the `routy` CLI call the same Rust core, so a request that works in the GUI
		works in CI.
	</Card>
	<Card title="Secrets stay out of the repo" icon="padlock">
		Tokens go to the system keychain, or come from `ROUTY_*` environment variables in CI. Nothing to
		accidentally commit.
	</Card>
	<Card title="Chains and checks" icon="approve-check">
		`> save` a value from one response, use it in the next request, `> assert` on status, headers and
		JSON body. Non-zero exit code when something fails.
	</Card>
</CardGrid>
