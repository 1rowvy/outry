import { Card, CardGrid } from '@astrojs/starlight/components';

```sh
curl -fsSL https://raw.githubusercontent.com/1rowvy/routy/master/install.sh | sh
```

```routy title="api/users/create.routy"
// Create user
POST /users {
  headers { Authorization: "Bearer ${Login().body.token}" }
  body { name: "Viktor", role: "admin" }

  expect {
    status == 201
    body matches { id: string, name: string }
  }

  save user_id = body.id
}
```

<CardGrid>
	<Card title="Requests live in git" icon="document">
		Requests are `.routy` files: JSON bodies, checks that read like code. Review them in pull requests,
		grep them, edit them in any editor — the app picks up changes instantly.
	</Card>
	<Card title="Same engine everywhere" icon="laptop">
		The desktop app and the `routy` CLI call the same Rust core, so a request that works in the GUI
		works in CI.
	</Card>
	<Card title="Secrets stay out of the repo" icon="padlock">
		Tokens go to the system keychain, or come from `ROUTY_*` environment variables in CI. Nothing to
		accidentally commit.
	</Card>
	<Card title="Requests call requests" icon="approve-check">
		`Login().body.token` sends the login once per run and reuses it; `flow` describes a scenario.
		`expect` checks status, headers and body shape. Non-zero exit code when something fails.
	</Card>
</CardGrid>
